cmake_minimum_required(VERSION 3.24)
project(roadpilot_scs_plugin LANGUAGES CXX)

if(NOT WIN32)
  message(FATAL_ERROR "The SCS plugin transport currently targets Windows named pipes.")
endif()

option(ROADPILOT_BUILD_PLUGIN "Build the SCS plugin DLL (requires SCS_SDK_ROOT)" ON)
option(ROADPILOT_BUILD_TESTS "Build the serializer unit tests (no SDK required)" ON)

# SDK-independent frame serializer, shared by the plugin and the tests.
add_library(roadpilot_frame STATIC src/frame_v2.cpp)
target_compile_features(roadpilot_frame PUBLIC cxx_std_17)
target_include_directories(roadpilot_frame PUBLIC src)
if(MSVC)
  target_compile_options(roadpilot_frame PRIVATE /W4 /permissive- /utf-8)
endif()

if(ROADPILOT_BUILD_PLUGIN)
  if(NOT DEFINED SCS_SDK_ROOT OR NOT EXISTS "${SCS_SDK_ROOT}/include/scssdk_telemetry.h")
    message(FATAL_ERROR "Set SCS_SDK_ROOT to an extracted SCS Telemetry SDK directory, or configure with -DROADPILOT_BUILD_PLUGIN=OFF.")
  endif()
  add_library(roadpilot_telemetry_plugin SHARED src/plugin.cpp src/roadpilot_telemetry_plugin.def)
  target_include_directories(roadpilot_telemetry_plugin PRIVATE "${SCS_SDK_ROOT}/include" "${SCS_SDK_ROOT}/include/common" "${SCS_SDK_ROOT}/include/eurotrucks2" "${SCS_SDK_ROOT}/include/amtrucks")
  target_compile_definitions(roadpilot_telemetry_plugin PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(roadpilot_telemetry_plugin PRIVATE roadpilot_frame advapi32)
  if(MSVC)
    target_compile_options(roadpilot_telemetry_plugin PRIVATE /W4 /permissive- /utf-8)
  endif()
  set_target_properties(roadpilot_telemetry_plugin PROPERTIES OUTPUT_NAME "roadpilot-telemetry")
endif()

if(ROADPILOT_BUILD_TESTS)
  enable_testing()
  add_executable(roadpilot_frame_tests tests/frame_v2_tests.cpp)
  target_link_libraries(roadpilot_frame_tests PRIVATE roadpilot_frame)
  target_compile_definitions(roadpilot_frame_tests PRIVATE ROADPILOT_FIXTURES_DIR="${CMAKE_CURRENT_SOURCE_DIR}/../schemas/fixtures")
  if(MSVC)
    target_compile_options(roadpilot_frame_tests PRIVATE /W4 /permissive- /utf-8)
  endif()
  add_test(NAME roadpilot_frame_tests COMMAND roadpilot_frame_tests)
endif()

# Loads the built DLL as a fake SCS host and talks to it over the real pipe. Needs the SDK
# headers and no running RoadPilot or telemetry reader.
if(ROADPILOT_BUILD_PLUGIN AND ROADPILOT_BUILD_TESTS)
  add_executable(roadpilot_plugin_host_tests tests/plugin_host_tests.cpp)
  target_compile_features(roadpilot_plugin_host_tests PRIVATE cxx_std_17)
  target_include_directories(roadpilot_plugin_host_tests PRIVATE "${SCS_SDK_ROOT}/include" "${SCS_SDK_ROOT}/include/common" "${SCS_SDK_ROOT}/include/eurotrucks2" "${SCS_SDK_ROOT}/include/amtrucks")
  target_compile_definitions(roadpilot_plugin_host_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX ROADPILOT_PLUGIN_PATH="$<TARGET_FILE:roadpilot_telemetry_plugin>")
  if(MSVC)
    target_compile_options(roadpilot_plugin_host_tests PRIVATE /W4 /permissive- /utf-8)
  endif()
  add_dependencies(roadpilot_plugin_host_tests roadpilot_telemetry_plugin)
  add_test(NAME roadpilot_plugin_host_tests COMMAND roadpilot_plugin_host_tests)
  set_tests_properties(roadpilot_plugin_host_tests PROPERTIES TIMEOUT 60)
endif()
